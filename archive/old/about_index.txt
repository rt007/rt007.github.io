<script>
  let currentSlide = 0;
  const slides = document.querySelectorAll('.carousel-item');
  const totalSlides = slides.length;
  const SLIDE_DURATION = 20000; // 20 seconds
  let slideInterval;
  let fillAnimationFrame;
  let fillStartTime;

  // All pagination bar sets (one per slide)
  // Each slide has its own set of bars; we control fill via JS on the active bar of the current slide
  // We collect all "bar-N" active fills dynamically from the current active slide
  function getActiveFill() {
    // Find the active slide's active bar fill
    const activeSlide = slides[currentSlide];
    const activePaginationBar = activeSlide.querySelector('.pagination-bar.active .pagination-bar-fill');
    return activePaginationBar;
  }

  function updatePaginationBars(index) {
    slides.forEach((slide, i) => {
      const bars = slide.querySelectorAll('.pagination-bar');
      const fills = slide.querySelectorAll('.pagination-bar-fill');
      bars.forEach((bar, j) => {
        bar.classList.toggle('active', j === index);
      });
      fills.forEach((fill, j) => {
        fill.style.transition = 'none';
        fill.style.width = j < index ? '100%' : '0%';
      });
    });
  }

  function startFillAnimation() {
    cancelAnimationFrame(fillAnimationFrame);
    fillStartTime = null;

    const fill = getActiveFill();
    if (!fill) return;
    fill.style.transition = 'none';
    fill.style.width = '0%';

    requestAnimationFrame(() => {
      requestAnimationFrame(() => {
        fill.style.transition = `width ${SLIDE_DURATION}ms linear`;
        fill.style.width = '100%';
      });
    });
  }

  function showSlide(index) {
    slides.forEach(slide => slide.classList.remove('active'));
    slides[index].classList.add('active');
    currentSlide = index;
    updatePaginationBars(index);
    startFillAnimation();
  }

  function nextSlide() {
    let next = (currentSlide + 1) % totalSlides;
    showSlide(next);
    resetTimer();
  }

  function prevSlide() {
    let prev = (currentSlide - 1 + totalSlides) % totalSlides;
    showSlide(prev);
    resetTimer();
  }

  function goToSlide(index) {
    showSlide(index);
    resetTimer();
  }

  function startTimer() {
    slideInterval = setInterval(nextSlide, SLIDE_DURATION);
  }

  function resetTimer() {
    clearInterval(slideInterval);
    startTimer();
  }

  // Initialize
  showSlide(0);
  startTimer();
</script>
